Escopo de recomposição

(Recomposition Scope)

@Composable fun Screen() { var value by remember { mutableStateOf(0) } Row( verticalAlignment = Alignment.CenterVertically, horizontalArrangement = Arrangement.Center, modifier = Modifier .fillMaxSize() ){ ButtonNotification( { value++ } ) MessageNotification(value) } }
@Composable fun ButtonNotification(increment: () -> Unit){ Button( onClick = { increment() } ) { Text("Agora vai") } }
@Composable fun MessageNotification(numero: Int){ Row(verticalAlignment = Alignment.CenterVertically){ Icon( imageVector = Icons.Filled.Person, contentDescription = null ) Text(text= "O valor é: $numero" ) } }
