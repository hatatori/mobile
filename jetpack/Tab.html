@Composable fun SimpleTabs() { var selectedTab by remember { mutableStateOf(0) } Column(modifier = Modifier.fillMaxSize()) { PrimaryTabRow(selectedTabIndex = selectedTab) { Tab( selected = selectedTab == 0, onClick = { selectedTab = 0 }, text = { Text("Aba 1") } ) Tab( selected = selectedTab == 1, onClick = { selectedTab = 1 }, text = { Text("Aba 2") } ) Tab( selected = selectedTab == 2, onClick = { selectedTab = 2 }, text = { Text("Aba 3") } ) } when (selectedTab) { 0 -> Text("Conteúdo da Aba 1") 1 -> Text("Conteúdo da Aba 2") 2 -> Text("Conteúdo da Aba 3") } } }
@Composable fun SimpleTabsForEach() { var selectedTab by remember { mutableStateOf(0) } val titles = listOf("Aba 1", "Aba 2", "Aba 3") Column(modifier = Modifier.fillMaxSize()) { PrimaryTabRow(selectedTabIndex = selectedTab) { titles.forEachIndexed { index, title -> Tab( selected = selectedTab == index, onClick = { selectedTab = index }, text = { Text(title) } ) } } when (selectedTab) { 0 -> Text("Conteúdo da Aba 1") 1 -> Text("Conteúdo da Aba 2") 2 -> Text("Conteúdo da Aba 3") } } }
