implementation("com.squareup.retrofit2:retrofit:3.0.0") implementation("com.squareup.retrofit2:converter-gson:3.0.0")
data class User( val id: Int? = null, val name: String? = null, val email: String, val message: String, ){}
interface ApiService { @GET("users") suspend fun getUsers(): List @POST("users") suspend fun addUser(@Body user: User): User }
object RetrofitClient { val api: ApiService by lazy { Retrofit.Builder() .baseUrl("https://SUA_API_DATABASE.mockapi.io/") .addConverterFactory(GsonConverterFactory.create()) .build() .create(ApiService::class.java) } }
fun ApiGet(){ scope.launch { val result = RetrofitClient.api.getUsers() users.addAll(result) } } fun ApiPost(){ scope.launch { val u = User( name = "maria", email = "maria@gmail.com", message = "message maria", ) RetrofitClient.api.addUser(u) } }
val users = remember { mutableStateListOf() } val scope = rememberCoroutineScope() Column{ Button( onClick = { ApiGet() } ){ Text(text="GET") } Button( onClick = { ApiPost() } ){ Text(text="POST") } LazyColumn { items(users) { user -> Text(text = user.name ?: "") } } }
